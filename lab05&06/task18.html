6. Code Block

// Theme detection with CSS
@media (prefers-color-scheme: dark) {
    :root {
        --bg-color: #1a1a1a;
        --text-color: #ffffff;
    }
}

@media (prefers-contrast: high) {
    * {
        border: 2px solid currentColor;
    }
}

@media (prefers-reduced-motion: reduce) {
    * {
        animation: none;
        transition: none;
    }
}
